        [1] = {
                .iomem = __UDIF_IOMEM_INIT(SIO_NAME "1", SIO_IO_BASE(1), SIO_IO_SIZE),
                .intr = UDIF_INTRPT_INIT(SIO_NAME "1", SIO_IRQ(1), 0),

        [2] = {
                .iomem = __UDIF_IOMEM_INIT(SIO_NAME "2", SIO_IO_BASE(2), SIO_IO_SIZE),

        },
