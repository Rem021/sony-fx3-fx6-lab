                    // 导航高亮逻辑

                    var navLinks = $(".navbar a");
                    var currentIndex = -1;

                    // 遍历所有sectiontitles，找出当前应该高亮的导航项
                    for (var i = 0; i < sectionTitles.length; i++) {
                        if (scrollTop > $(sectionTitles[i]).offset().top - 121) {
                            currentIndex = i;
                        }
                    }

                    // 更新导航高亮状态
                    navLinks.removeClass("current");
                    if (currentIndex >= 0 && currentIndex < navLinks.length) {
                        navLinks.eq(currentIndex).addClass("current");
                    }

                    // 处理navtbg和duibi_open的显示逻辑
                    var section9999 = $(".section9999");
                    if (section9999.length) {
                        var showSlider = scrollTop > section9999.offset().top - 121;
                        $(".navtbg").toggleClass("slidershow", showSlider);
                    }

                } catch (e) {
                    console.error("Scroll handler error:", e);
                }
            })
